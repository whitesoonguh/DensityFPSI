include(CheckCCompilerFlag)

add_library(gf64poly src/gf64_poly.c src/gf64_lin.c)
target_include_directories(gf64poly PUBLIC include)
add_library(gf64::gf64 ALIAS gf64poly)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gf64poly PUBLIC -mpclmul -msse4.1)
endif()

add_executable(test_gf64_poly test/test_gf64_poly.c)
target_link_libraries(test_gf64_poly PRIVATE gf64poly)
add_test(NAME gf64_scalar COMMAND test_gf64_poly)

set(AVX512_FLAGS -mpclmul -msse4.1 -mavx512f -mavx512bw -mvpclmulqdq)
set(CMAKE_REQUIRED_FLAGS "${AVX512_FLAGS}")
check_c_compiler_flag("" HAVE_VPCLMULQDQ_FLAGS)
unset(CMAKE_REQUIRED_FLAGS)

if(NOT DEFINED FPSI_ENABLE_AVX512)
  option(FPSI_ENABLE_AVX512 "Enable AVX-512/VPCLMUL code paths" ON)
endif()
set(GF64_ENABLE_AVX512_DEFAULT OFF)
if(FPSI_ENABLE_AVX512 AND HAVE_VPCLMULQDQ_FLAGS)
  set(GF64_ENABLE_AVX512_DEFAULT ON)
endif()
option(GF64_ENABLE_AVX512 "Build VPCLMULQDQ batch-4 module" ${GF64_ENABLE_AVX512_DEFAULT})

if(GF64_ENABLE_AVX512 AND HAVE_VPCLMULQDQ_FLAGS)
  add_library(gf64x4poly src/gf64x4_poly.c src/gf64x4_gcd.c src/gf64x4_linalg.c)
  add_library(gf64::gf64x4 ALIAS gf64x4poly)
  target_include_directories(gf64x4poly PUBLIC include)
  target_compile_options(gf64x4poly PUBLIC ${AVX512_FLAGS})

  add_executable(test_gf64x4_poly test/test_gf64x4_poly.c)
  target_link_libraries(test_gf64x4_poly PRIVATE gf64x4poly gf64poly)
  add_test(NAME gf64_batch4 COMMAND test_gf64x4_poly)

  add_executable(test_gf64x4_gcd test/test_gf64x4_gcd.c)
  target_link_libraries(test_gf64x4_gcd PRIVATE gf64x4poly gf64poly)
  add_test(NAME gf64_batch4_gcd COMMAND test_gf64x4_gcd)

  add_executable(bench_gf64_core bench/bench_gf64_core.c)
  target_link_libraries(bench_gf64_core PRIVATE gf64x4poly gf64poly)
  target_compile_definitions(bench_gf64_core PRIVATE GF64_BENCH_HAVE_X4=1)
  target_compile_options(bench_gf64_core PRIVATE ${AVX512_FLAGS})
  message(STATUS "gf64 batch-4 (VPCLMULQDQ): ENABLED (requires runtime VPCLMULQDQ)")
else()
  add_executable(bench_gf64_core bench/bench_gf64_core.c)
  target_link_libraries(bench_gf64_core PRIVATE gf64poly)
  message(STATUS "gf64 batch-4 (VPCLMULQDQ): disabled")
endif()
