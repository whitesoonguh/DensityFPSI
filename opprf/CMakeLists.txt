include(CheckCXXCompilerFlag)

if(NOT DEFINED FPSI_ENABLE_AVX512)
    option(FPSI_ENABLE_AVX512 "Enable AVX-512 code paths" ON)
endif()
option(SOOPRF_ENABLE_AVX512 "Compile AVX-512 code paths" ${FPSI_ENABLE_AVX512})
option(SOOPRF_BUILD_TESTS "Build tests" ON)
option(SOOPRF_BUILD_BENCH "Build local benchmarks" ON)

add_library(sooprf_altmod STATIC
    src/altmod_prf.cpp
    src/shared_oprf.cpp
)
add_library(sooprf::altmod ALIAS sooprf_altmod)
target_include_directories(sooprf_altmod PUBLIC include)
target_compile_features(sooprf_altmod PUBLIC cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(sooprf_altmod PRIVATE -O3 -funroll-loops -Wall -Wextra -Wpedantic)
    if(FPSI_ENABLE_NATIVE)
        target_compile_options(sooprf_altmod PRIVATE -march=native)
    endif()
    if(NOT SOOPRF_ENABLE_AVX512)
        target_compile_options(sooprf_altmod PRIVATE -mno-avx512f -mno-avx512bw -mno-avx512vl -mno-vaes)
    endif()
endif()
if(SOOPRF_ENABLE_AVX512)
    check_cxx_compiler_flag("-mavx512f" HAS_AVX512F)
    check_cxx_compiler_flag("-mavx512bw" HAS_AVX512BW)
    check_cxx_compiler_flag("-mavx512vl" HAS_AVX512VL)
    if(HAS_AVX512F AND HAS_AVX512BW AND HAS_AVX512VL)
        target_compile_options(sooprf_altmod PRIVATE -mavx512f -mavx512bw -mavx512vl)
        target_compile_definitions(sooprf_altmod PUBLIC SOOPRF_COMPILED_AVX512=1)
    endif()
endif()

add_library(opprf STATIC src/opprf.cpp)
add_library(opprf::opprf ALIAS opprf)
target_include_directories(opprf PUBLIC include)
target_compile_features(opprf PUBLIC cxx_std_20)
target_link_libraries(opprf PUBLIC sooprf::altmod okvs::okvs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(opprf PRIVATE -O3 -Wall -Wextra -Wpedantic)
    if(FPSI_ENABLE_NATIVE)
        target_compile_options(opprf PRIVATE -march=native)
    endif()
    if(NOT SOOPRF_ENABLE_AVX512)
        target_compile_options(opprf PRIVATE -mno-avx512f -mno-avx512bw -mno-avx512vl -mno-vaes)
    endif()
endif()

if(SOOPRF_BUILD_TESTS)
    add_executable(test_altmod test/test_altmod.cpp)
    target_link_libraries(test_altmod PRIVATE sooprf_altmod)
    add_test(NAME sooprf_altmod COMMAND test_altmod)

    add_executable(test_opprf test/test_opprf.cpp)
    target_link_libraries(test_opprf PRIVATE opprf::opprf)
    add_test(NAME opprf64 COMMAND test_opprf)

    add_executable(test_shared_oprf_ot_tape test/test_shared_oprf_ot_tape.cpp)
    target_link_libraries(test_shared_oprf_ot_tape PRIVATE sooprf_altmod)
    add_test(NAME sooprf_altmod_wprf_tape COMMAND test_shared_oprf_ot_tape)
endif()

if(SOOPRF_BUILD_BENCH)
    add_executable(bench_opprf bench/bench_opprf.cpp)
    target_link_libraries(bench_opprf PRIVATE opprf::opprf)
endif()
