cmake_minimum_required(VERSION 3.16)
project(okvs C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CheckCCompilerFlag)

if(NOT DEFINED FPSI_ENABLE_AVX512)
  option(FPSI_ENABLE_AVX512 "Enable AVX-512/VAES code paths" ON)
endif()
if(NOT DEFINED FPSI_ENABLE_AES)
  option(FPSI_ENABLE_AES "Enable AES-NI/VAES code paths" ON)
endif()

option(OKVS_ENABLE_AVX512 "Enable AVX-512 wide-XOR and VAES-512 OKVS paths" ${FPSI_ENABLE_AVX512})
option(OKVS_ENABLE_AES "Enable AES-NI/VAES OKVS PRF paths" ${FPSI_ENABLE_AES})

add_library(okvs src/okvs_w.c src/okvs_util.c src/okvs_encode.c)
target_include_directories(okvs PUBLIC include)
add_library(okvs::okvs ALIAS okvs)

set(AVX512_XOR -mavx512f -mavx512bw)
set(CMAKE_REQUIRED_FLAGS "${AVX512_XOR}")
check_c_compiler_flag("" HAVE_AVX512_XOR)
unset(CMAKE_REQUIRED_FLAGS)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND OKVS_ENABLE_AVX512 AND HAVE_AVX512_XOR)
  target_compile_options(okvs PRIVATE ${AVX512_XOR})
  message(STATUS "okvs array-V SIMD: AVX-512 wide XOR")
else()
  message(STATUS "okvs array-V SIMD: scalar fallback")
endif()

set(OKVS_VAES512_FLAGS -mavx512f -mavx512bw -mvaes -maes -msse4.1)
set(CMAKE_REQUIRED_FLAGS "${OKVS_VAES512_FLAGS}")
check_c_compiler_flag("" HAVE_OKVS_VAES512_FLAGS)
unset(CMAKE_REQUIRED_FLAGS)
set(OKVS_AESNI_FLAGS -maes -msse4.1)
set(CMAKE_REQUIRED_FLAGS "${OKVS_AESNI_FLAGS}")
check_c_compiler_flag("" HAVE_OKVS_AESNI_FLAGS)
unset(CMAKE_REQUIRED_FLAGS)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND OKVS_ENABLE_AVX512 AND OKVS_ENABLE_AES AND HAVE_OKVS_VAES512_FLAGS)
  target_compile_options(okvs PRIVATE ${OKVS_VAES512_FLAGS})
  target_compile_definitions(okvs PRIVATE OKVS_AES_IMPL_VAES512=1)
  message(STATUS "okvs band-derivation PRF: VAES-512")
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND OKVS_ENABLE_AES AND HAVE_OKVS_AESNI_FLAGS)
  target_compile_options(okvs PRIVATE ${OKVS_AESNI_FLAGS})
  target_compile_definitions(okvs PRIVATE OKVS_AES_IMPL_AESNI=1)
  message(STATUS "okvs band-derivation PRF: AES-NI fallback")
else()
  target_compile_definitions(okvs PRIVATE OKVS_AES_IMPL_SOFTWARE=1)
  message(STATUS "okvs band-derivation PRF: software AES fallback")
endif()

add_executable(test_okvs_w test/test_okvs_w.c)
target_link_libraries(test_okvs_w PRIVATE okvs)
add_test(NAME okvs_w COMMAND test_okvs_w)

add_executable(test_okvs_util test/test_okvs_util.c)
target_link_libraries(test_okvs_util PRIVATE okvs)
add_test(NAME okvs_util COMMAND test_okvs_util)

add_executable(test_okvs_encode test/test_okvs_encode.c)
target_link_libraries(test_okvs_encode PRIVATE okvs)
add_test(NAME okvs_encode COMMAND test_okvs_encode)

add_executable(test_okvs_decode_cache test/test_okvs_decode_cache.c)
target_link_libraries(test_okvs_decode_cache PRIVATE okvs)
add_test(NAME okvs_decode_cache COMMAND test_okvs_decode_cache)

add_executable(bench_okvs_core bench/bench_okvs_core.c)
target_link_libraries(bench_okvs_core PRIVATE okvs)
